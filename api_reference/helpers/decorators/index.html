    Decorators¶

    The helpers/decorators subpackage provides function and class decorators for cross-cutting concerns including caching, retry logic, deprecation, timing, tracing, and transaction management.

    Decorators are named without a _decorator suffix (retry, timeout, ttl_cache, ...) and detect async def functions automatically. The old names (retry_decorator, async_trace_span, ...) keep working and emit a DeprecationWarning and will be removed in 6.0.0; see the Decorators tutorial.

    class CachedFunction[**P, R]:
    """Wrapper class for a cached function with a clear_cache method.

    This class wraps a function to provide TTL-based caching. The cache is shared
    across all instances when used as an instance method decorator.

    Example:
        ```python
        @ttl_cache(ttl_seconds=60, maxsize=100)
        def expensive_function(x: int) -> int:
            return x * 2


        # First call executes the function
        result = expensive_function(5)  # Returns 10

        # Second call returns cached result
        result = expensive_function(5)  # Returns 10 (from cache)

        # Clear cache manually
        expensive_function.clear_cache()
        ```
    """

    def __init__(
        self,
        func: Callable[..., R],
        cache: MutableMapping[Hashable, object],
        instance: object | None = None,
        lock: threading.RLock | None = None,
        cache_factory: Callable[[], MutableMapping[Hashable, object]] | None = None,
        copy_results: bool = False,
    ) -> None:
        """Initialize the cached function wrapper.

        Args:
            func: The function to wrap.
            cache: The cache instance to use.
            instance: The instance this method is bound to (for bound methods).
            lock: Lock guarding the cache; shared between bound copies of the same decorated function.
            cache_factory: When set, every instance the method is bound to gets its own cache built by
                this factory (per-instance caching) instead of sharing ``cache``.
            copy_results: Return a deep copy of every result so callers mutating it cannot corrupt the cache.
        """
        self._func: Callable[..., R] = func
        self._cache: MutableMapping[Hashable, object] = cache
        self._instance: object | None = instance
        self._cache_factory = cache_factory
        self._copy_results = copy_results
        # Calls on the same cache from several threads must not race on the (non thread-safe) TTLCache
        self._lock = lock if lock is not None else threading.RLock()
        # Preserve function metadata
        wraps(func)(self)
        if inspect.iscoroutinefunction(func):
            inspect.markcoroutinefunction(self)

    def __get__(self, obj: object, objtype: type | None = None) -> CachedFunction[P, R]:
        """Support instance methods by implementing descriptor protocol.

        This method caches the bound method in the instance's __dict__ to ensure
        identity consistency (obj.method is obj.method returns True).
        """
        if obj is None:
            return self

        # Cache the bound method in the instance's __dict__ for identity consistency
        func_name = getattr(self._func, "__name__", "cached_method")
        # id(self) keeps same-named cached methods of different classes in one MRO apart
        bound_method_name = f"_cached_{func_name}_{id(self)}"
        if not hasattr(obj, bound_method_name):
            bound_cached = self._bind(obj)
            # Store in instance __dict__ to maintain identity
            try:
                object.__setattr__(obj, bound_method_name, bound_cached)
            except AttributeError, TypeError:
                # If we can't set the attribute (frozen dataclass, etc.), return the new bound copy uncached by identity
                return bound_cached

        return getattr(obj, bound_method_name)

    def _bind(self, obj: object) -> CachedFunction[P, R]:
        """Create the copy bound to ``obj``: shared cache by default, its own cache in per-instance mode."""
        if self._cache_factory is None:
            return CachedFunction(
                self._func,
                self._cache,
                instance=obj,
                lock=self._lock,
                copy_results=self._copy_results,
            )
        return CachedFunction(
            self._func,
            self._cache_factory(),
            instance=obj,
            lock=threading.RLock(),
            copy_results=self._copy_results,
        )

    def _emit(self, value: object) -> object:
        """Return ``value``, or a deep copy of it when results are copied."""
        return copy.deepcopy(value) if self._copy_results else value

    def _make_key(self, args: tuple[object, ...], kwargs: Mapping[str, object]) -> Hashable | None:
        """Build a hashable key from the call arguments, or None if they cannot be hashed."""
        from cachetools.keys import typedkey

        key = typedkey(*args, **kwargs)
        try:
            hash(key)
        except TypeError:
            return None
        return key

    def __call__(self, *args: object, **kwargs: object) -> R:
        """Call the cached function.

        Calls whose arguments are not hashable bypass the cache instead of failing.

        Args:
            *args: Positional arguments to pass to the function.
            **kwargs: Keyword arguments to pass to the function.

        Returns:
            The result of the function call (from cache or fresh). For ``async def`` functions
            the result is an awaitable and the awaited value is what gets cached.
        """
        key = self._make_key(args, kwargs)
        call_args = args if self._instance is None else (self._instance, *args)

        if inspect.iscoroutinefunction(self._func):
            # R is the coroutine type here; the cached coroutine wrapper stands in for it
            return cast("R", self._call_async(key, call_args, kwargs))

        if key is None:
            return self._func(*call_args, **kwargs)
        with self._lock:
            cached = self._cache.get(key, _MISSING)
        if cached is not _MISSING:
            return cast("R", self._emit(cached))

        result: R = self._func(*call_args, **kwargs)
        with self._lock:
            self._cache[key] = result
        return cast("R", self._emit(result))

    async def _call_async(
        self,
        key: Hashable | None,
        call_args: tuple[object, ...],
        kwargs: Mapping[str, object],
    ) -> object:
        """Await the wrapped coroutine function and cache its awaited value."""
        if key is None:
            return await cast("Awaitable[object]", self._func(*call_args, **kwargs))
        with self._lock:
            cached = self._cache.get(key, _MISSING)
        if cached is not _MISSING:
            return self._emit(cached)

        result = await cast("Awaitable[object]", self._func(*call_args, **kwargs))
        with self._lock:
            self._cache[key] = result
        return self._emit(result)

    def clear_cache(self) -> None:
        """Clear the cache.

        This clears all cached values for this function. When used with instance methods,
        this clears the shared cache for all instances, or only the bound instance's cache
        when the function was decorated with ``per_instance=True``.
        """
        with self._lock:
            self._cache.clear()

    def ttl_cache[**P, R](
    ttl_seconds: int | Callable[[], int] = 300,
    maxsize: int = 100,
    *,
    per_instance: bool = False,
    copy_results: bool = False,
) -> Callable[[Callable[P, R]], CachedFunction[P, R]]:
    """Decorator that provides a TTL cache for functions and methods.

    By default the cache is shared across all instances when decorating instance methods,
    which is only correct for results that depend on the method arguments alone. Pass
    ``per_instance=True`` when the result depends on instance state (endpoint, credentials, realm).

    Args:
        ttl_seconds: Time to live in seconds (default: 5 minutes), or a zero-argument
            callable that returns the TTL (evaluated once when the decorator is applied).
            After this time, cached entries expire and the function is re-executed.
        maxsize: Maximum size of the cache (default: 100).
            When the cache is full, the least recently used entry is evicted.
        per_instance: When True, every instance gets its own cache, so results never leak between
            instances and ``clear_cache()`` on one instance leaves the others untouched (default: False,
            one cache shared by all instances).
        copy_results: When True, callers receive a deep copy of the cached value, so mutating a returned
            list or dict cannot corrupt later cache hits (default: False, the cached object itself).

    Returns:
        Decorated function with TTL caching and a clear_cache() method.

    Raises:
        ConfigurationError: If ``cachetools`` is not installed (``archipy[cache]`` extra).

    Example:
        ```python
        class DataService:
            @ttl_cache(ttl_seconds=60, maxsize=50)
            def fetch_data(self, key: str) -> dict:
                # Expensive operation
                return {"data": key}


        service1 = DataService()
        service2 = DataService()

        # First call executes the function
        result1 = service1.fetch_data("key1")

        # Second call from different instance returns cached result
        result2 = service2.fetch_data("key1")  # From cache

        # Clear cache manually
        service1.fetch_data.clear_cache()
        ```

    Note:
        - Exceptions are not cached; the function will be re-executed on the next call
        - None values are cached like any other value
        - Argument types are part of the key (``f(1)`` and ``f(1.0)`` are cached separately) and
          arguments must be hashable; calls with unhashable arguments run uncached
        - ``async def`` functions are supported; the awaited result is cached
        - Access to the cache is guarded by a lock, so it is safe to call from several threads
        - Cache is shared across all instances of a class unless ``per_instance=True`` is passed
    """
    _require_cachetools()
    from cachetools import TTLCache

    resolved_ttl = int(ttl_seconds) if isinstance(ttl_seconds, int) else int(ttl_seconds())
    cache: TTLCache[Hashable, object, float] = TTLCache(maxsize=maxsize, ttl=resolved_ttl)
    lock = threading.RLock()

    def new_cache() -> MutableMapping[Hashable, object]:
        return TTLCache(maxsize=maxsize, ttl=resolved_ttl)

    def decorator(func: Callable[P, R]) -> CachedFunction[P, R]:
        return CachedFunction(
            func,
            cache,
            instance=None,
            lock=lock,
            cache_factory=new_cache if per_instance else None,
            copy_results=copy_results,
        )

    return decorator

    def retry[**P, R](
    max_retries: int = 3,
    delay: float = 1,
    retry_on: tuple[type[Exception], ...] | None = None,
    ignore: tuple[type[Exception], ...] | None = None,
    resource_type: str | None = None,
    backoff: float = 1.0,

) -> Callable[[Callable[P, R]], Callable[P, R]]:
    """A decorator that retries a function when it raises an exception.

    Works for both regular and ``async def`` functions. Async functions are awaited and
    wait with ``asyncio.sleep`` between attempts, so the event loop is never blocked.

    Args:

            (``3`` means one call plus two retries). Must be at least 1. Defaults to 3.

    def timeout[**P, R](seconds: float) -> Callable[[Callable[P, R]], Callable[P, R]]:
    """A decorator that adds a timeout to a function.

    If the function takes longer than the specified number of seconds to execute,
    a `DeadlineExceededError` is raised. The timeout is implemented with
    `signal.setitimer(signal.ITIMER_REAL, ...)`, so fractional seconds are supported.
    After the call, the previous SIGALRM handler is restored and any timer that was
    pending before the call is re-armed with its remaining time.

    ``async def`` functions are supported too: they are bounded with `asyncio.timeout`, which
    works in any thread and needs no signals.

    Limitations:
        Synchronous functions rely on `SIGALRM`, so they work only on Unix and only when the
        decorated function is called from the main thread of the main interpreter. Calling one
        from another thread raises `FailedPreconditionError` instead of running without a timeout.
        It cannot interrupt blocking C calls that do not return to the interpreter.

    Args:
        seconds (float): The maximum number of seconds the function is allowed to run.

    Returns:
        Callable: The decorated function with a timeout.

    Example:
        To use this decorator, apply it to any function and specify the timeout in seconds:

        ```python
        @timeout(3)  # Set a timeout of 3 seconds
        def long_running_function():
            time.sleep(5)  # This will take longer than the timeout
            return "Finished"


        try:
            result = long_running_function()
        except DeadlineExceededError as e:
            print(e.additional_data)  # {"timeout": 3, "operation": "long_running_function"}
        ```
    """

    def decorator(func: Callable[P, R]) -> Callable[P, R]:
        # Capture function name before wrapping - use getattr for type safety
        func_name = getattr(func, "__name__", "unknown")

        if inspect.iscoroutinefunction(func):
            # R is the coroutine type here; the wrapper awaits it and returns the awaited value
            async_func = cast("Callable[P, Awaitable[object]]", func)

            @wraps(func)
            async def async_wrapper(*args: P.args, **kwargs: P.kwargs) -> object:
                try:
                    async with asyncio.timeout(seconds):
                        return await async_func(*args, **kwargs)
                except TimeoutError as e:
                    raise DeadlineExceededError(timeout=seconds, operation=func_name) from e

            return cast("Callable[P, R]", async_wrapper)

        @wraps(func)
        def wrapper(*args: P.args, **kwargs: P.kwargs) -> R:
            if threading.current_thread() is not threading.main_thread():
                raise FailedPreconditionError(
                    precondition="timeout requires the main thread",
                    additional_data={"operation": func_name},
                )

            def handle_timeout(_signum: int, _frame: FrameType | None) -> None:
                raise DeadlineExceededError(timeout=seconds, operation=func_name)

            previous_handler = signal.signal(signal.SIGALRM, handle_timeout)
            started = time.monotonic()
            previous_remaining, previous_interval = signal.setitimer(signal.ITIMER_REAL, seconds)

            try:
                return func(*args, **kwargs)
            finally:
                # Disable our timer, restore the previous handler and any timer that was pending
                signal.setitimer(signal.ITIMER_REAL, 0)
                signal.signal(signal.SIGALRM, previous_handler)
                if previous_remaining > 0:
                    remaining = max(previous_remaining - (time.monotonic() - started), 1e-6)
                    signal.setitimer(signal.ITIMER_REAL, remaining, previous_interval)

        return wrapper

    return decorator

        for method_name, attr in list(vars(cls).items()):
            if method_name.startswith("_") or method_name in excluded:
                continue
            if isinstance(attr, property | type):
                continue

                wrapped = _wrap_method(cls, method_name, attr.__func__, capture_args=capture_args)
                setattr(cls, method_name, staticmethod(wrapped))
            elif isinstance(attr, classmethod):
                wrapped = _wrap_method(cls, method_name, attr.__func__, capture_args=capture_args)
                setattr(cls, method_name, classmethod(wrapped))
            elif callable(attr):
                setattr(cls, method_name, _wrap_method(cls, method_name, attr, capture_args=capture_args))

        return cls

    def singleton[T: type[Any]](*, thread_safe: bool = True) -> Callable[[T], T]:  # Any: patches __new__/__init__
    """A decorator to create thread-safe Singleton classes.

    This decorator ensures that only one instance of a class is created. The decorated name
    is still the class itself, so ``isinstance``, subclassing and class-level attributes keep
    working. Constructor arguments passed after the first instantiation are ignored.

    Args:
        thread_safe (bool, optional): If True, enables thread-safety for instance creation.
                                      Defaults to True.

    Returns:
        function: A decorator function that can be applied to a class.

    Example:
        To create a Singleton class, apply the `singleton` and optionally specify
        whether thread-safety should be enabled:

        ```python
        @singleton(thread_safe=True)
        class MySingletonClass:
            def __init__(self, value):
                self.value = value


        # Create instances of MySingletonClass
        instance1 = MySingletonClass(10)
        instance2 = MySingletonClass(20)

        # Verify that both instances are the same
        print(instance1.value)  # Output: 10
        print(instance2.value)  # Output: 10
        print(instance1 is instance2)  # Output: True
        print(isinstance(instance1, MySingletonClass))  # Output: True
        ```
    """

    def decorator(cls: T) -> T:
        """Patch the class so that it only ever builds and initializes one instance.

        Args:
            cls: The class to be decorated as a Singleton.

        Returns:
            The same class, now limited to a single instance per (sub)class.
        """
        instances: dict[type, object] = {}
        initialized: set[type] = set()
        # Without thread safety a no-op context manager replaces the lock
        lock: AbstractContextManager[object] = threading.RLock() if thread_safe else nullcontext()
        original_new: Callable[..., object] = cls.__new__
        original_init: Callable[..., None] = cls.__init__

        def singleton_new(klass: type, *args: object, **kwargs: object) -> object:
            if klass not in instances:
                with lock:
                    if klass not in instances:
                        if original_new is object.__new__:
                            instances[klass] = original_new(klass)
                        else:
                            instances[klass] = original_new(klass, *args, **kwargs)
            return instances[klass]

        def singleton_init(self: object, *args: object, **kwargs: object) -> None:
            klass = type(self)
            if klass not in initialized:
                with lock:
                    if klass not in initialized:
                        original_init(self, *args, **kwargs)
                        initialized.add(klass)

        cls.__new__ = singleton_new  # type: ignore[assignment]
        cls.__init__ = singleton_init  # type: ignore[misc]
        return cls

    return decorator

    def class_deprecation_error[T: type[Any]](  # Any: patches __init__ on the class
    operation: str | None = None,
    lang: LanguageType = LanguageType.FA,
) -> Callable[[T], T]:
    """A decorator that raises a DeprecationException when the decorated class is instantiated.

    Args:
        operation (str, optional): The name of the operation that is deprecated.
            Defaults to the name of the decorated class.
        lang (str): The language for the error message (default: "fa").

    Returns:
        Callable: The decorated class that raises a DeprecationException.

    Example:
        To use this decorator, apply it to a class:

        ```python
        @class_deprecation_error(operation="OldClass", lang="en")
        class OldClass:
            def __init__(self):
                pass


        # Instantiating the class will raise a DeprecationException
        obj = OldClass()
        ```

        Output:
        ```
        DeprecationException: This operation is deprecated and will be removed in a future version.
        Operation: OldClass
        ```

    def decorator(cls: T) -> T:
        @wraps(cls.__init__)
        def new_init(_self: object, *_args: object, **_kwargs: object) -> None:
            operation_name = operation if operation is not None else cls.__name__
            raise DeprecationError(deprecated_feature=operation_name, lang=lang)

        cls.__init__ = new_init
        return cls

    return decorator

    def grpc_rate_limit[**P, R](
    calls_count: int = 1,
    milliseconds: int = 0,
    seconds: int = 0,
    minutes: int = 0,
    hours: int = 0,
    days: int = 0,
) -> Callable[[Callable[P, R]], Callable[P, R]]:
    """Attach a rate-limit window to a gRPC servicer method without wrapping it.

    Stack multiple decorators on one method to enforce burst and sustained limits.
    The ``GrpcServerRateLimitInterceptor`` reads ``__archipy_rate_limit_windows__``
    and enforces each window in declaration order.

    Args:
        calls_count: Maximum allowed requests within the window.
        milliseconds: Milliseconds component of the window.
        seconds: Seconds component of the window.
        minutes: Minutes component of the window.
        hours: Hours component of the window.
        days: Days component of the window.

    Returns:
        A decorator that records the window on the target function.

    Raises:
        InvalidArgumentError: If ``calls_count`` or the computed window is invalid.
    """

    def decorator(func: Callable[P, R]) -> Callable[P, R]:
        window = RateLimitUtils.compute_rate_limit_window(
            calls_count=calls_count,
            milliseconds=milliseconds,
            seconds=seconds,
            minutes=minutes,
            hours=hours,
            days=days,
        )
        existing = getattr(func, RateLimitUtils.ARCHIPY_RATE_LIMIT_WINDOWS_ATTR, ())
        setattr(func, RateLimitUtils.ARCHIPY_RATE_LIMIT_WINDOWS_ATTR, (*existing, window))
        return func

    return decorator
